## The Core Insight

Users don't want features.

Users want outcomes.

### What Users Actually Say

- "I want to build a startup" (not "I want a project management tool")
- "I want to write a book" (not "I want a document editor")
- "I want to organize my team" (not "I want a CRM")
- "I want to learn AI" (not "I want a note-taking app")
- "I want to find a job" (not "I want a task tracker")

### What Matters

The user expresses an **outcome**.

Diana understands the outcome is a **mission**.

Diana transforms the mission into a **workspace** with everything needed to succeed.

### The Principle

> **Users start missions. Diana assembles capabilities.**

### Layer 1: Intent Recognition

```
User Input:
  "I want to build a startup"
  
Intent Recognition:
  Type: Mission
  Domain: Business/Entrepreneurship
  Scope: Long-term (weeks/months)
  Category: Startup
  
Confidence: 0.98
Alternative Matches: [Build SaaS, Launch Product]

```
Mission Classification:
  ID: startup-v1
  Name: "Build a Startup"
  Description: "Create a complete business from idea to launch"
  Estimated Duration: 12-16 weeks
  Complexity: High
  Teams Required: Yes
  
Domain Knowledge:
  - Business planning
  - Market research
  - Financial modeling

```

```

model Mission {
  id                 String                @id @default(cuid())
  userId             String

  // Mission metadata
  missionTemplateId  String
  missionTemplate    MissionTemplate       @relation(fields: [missionTemplateId], references: [id])
  
  // Instance data
  name               String
  description        String?
  status             MissionStatus         @default(ACTIVE) // ACTIVE, PAUSED, COMPLETED, ARCHIVED
  startDate          DateTime              @default(now())
  targetDate         DateTime?
  completionDate     DateTime?
  
  // Context
  workspaceId        String?
  workspace          Workspace?            @relation(fields: [workspaceId], references: [id])
  
  // Progress
  phase              String                @default("setup")
  progress           Int                   @default(0) // 0-100
  
  // Configuration
  customization      Json?                 // Store user preferences for template
  
  // Relationships
  documents          Document[]
  tasks              ProjectTask[]
  memories           DianaMemory[]
  updates            MissionUpdate[]
  
  createdAt          DateTime              @default(now())
  updatedAt          DateTime              @updatedAt
}

model MissionTemplate {
  id                 String                @id @default(cuid())
  name               String
  slug               String                @unique
  description        String
  category           String                // startup, learning, business, writing, productivity
  
  // Template structure
  documents          TemplateDocument[]
  taskTemplates      MissionTaskTemplate[]
  phases             MissionPhase[]
  
  // Metadata
  difficulty         String                // beginner, intermediate, advanced
  estimatedDuration  Int                   // days
  minTeamSize        Int
  
  createdAt          DateTime              @default(now())
  updatedAt          DateTime              @updatedAt
}

model MissionPhase {
  id                 String                @id @default(cuid())
  missionTemplateId  String
  missionTemplate    MissionTemplate       @relation(fields: [missionTemplateId], references: [id])
  
  name               String
  description        String
  order              Int
  estimatedDuration  Int                   // days
  
  // Conditions to advance
  tasksRequired      String[]              // Task IDs that must complete
  checkpointChecklist Json?                // Checklist for this phase
  
  createdAt          DateTime              @default(now())
}

model MissionUpdate {
  id                 String                @id @default(cuid())
  missionId          String
  mission            Mission               @relation(fields: [missionId], references: [id])
  
  type               String                // phase_advanced, task_completed, milestone_achieved
  title              String
  description        String?
  
  createdAt          DateTime              @default(now())
}

enum MissionStatus {
  ACTIVE
  PAUSED
  COMPLETED
  ARCHIVED
}

## The Mission Engine Service

```typescript
// apps/api/src/missions/mission.service.ts

@Injectable()
export class MissionService {
  constructor(
    private prisma: PrismaService,
    private intentRecognition: IntentRecognitionService,
    private orchestrator: WorkspaceOrchestratorService,
    private diana: DianaService
  ) {}

  // 1. Recognize intent and suggest missions
  async suggestMissions(userInput: string): Promise<MissionSuggestion[]> {
    const intent = await this.intentRecognition.analyze(userInput);
    
    if (intent.type !== 'MISSION') {
      return [];
    }
    
    // Match against templates
    const suggestions = await this.prisma.missionTemplate
      .findMany({
        where: {
          category: { in: intent.categories }
        },
        orderBy: { relevance: 'desc' },
        take: 3
      });
    
    return suggestions.map(t => ({
      templateId: t.id,
      name: t.name,
      description: t.description,
      confidence: this.calculateConfidence(intent, t),
      estimatedDuration: t.estimatedDuration
    }));
  }

  // 2. Launch a mission
  async launchMission(
    userId: string,
    missionTemplateName: string,
    customization?: Record<string, any>
  ): Promise<Mission> {
    // Get template
    const template = await this.prisma.missionTemplate.findUnique({
      where: { slug: missionTemplateName },
      include: { documents: true, phases: true }
    });

    // Create workspace via Orchestrator
    const workspace = await this.orchestrator.createWorkspace({
      name: template.name,
      template: template.slug,
      customization
    });

    // Create mission record
    const mission = await this.prisma.mission.create({
      data: {
        userId,
        missionTemplateId: template.id,
        name: template.name,
        workspaceId: workspace.id,
        customization
      }
    });

    // Initialize memory profile
    await this.diana.initMissionMemory(userId, mission.id, template);

    // Send Diana welcome message
    await this.diana.sendMissionWelcome(userId, mission);

    return mission;
  }

  // 3. Get mission progress
  async getMissionProgress(missionId: string) {
    const mission = await this.prisma.mission.findUnique({
      where: { id: missionId },
      include: {
        tasks: { where: { status: 'done' } },
        documents: { select: { id: true, updatedAt: true } },
        updates: { orderBy: { createdAt: 'desc' }, take: 10 }
      }
    });

    const totalTasks = await this.prisma.projectTask.count({
      where: { projectId: mission.workspaceId }
    });

    return {
      mission,
      progress: Math.round((mission.tasks.length / totalTasks) * 100),
      recentUpdates: mission.updates
    };
  }

  // 4. Advance mission phase
  async advancePhase(missionId: string): Promise<void> {
    const mission = await this.prisma.mission.findUnique({
      where: { id: missionId }
    });

    const template = await this.prisma.missionTemplate.findUnique({
      where: { id: mission.missionTemplateId },
      include: { phases: { orderBy: { order: 'asc' } } }
    });

    const currentPhaseIndex = template.phases.findIndex(
      p => p.name === mission.phase
    );
    
    if (currentPhaseIndex < template.phases.length - 1) {
      const nextPhase = template.phases[currentPhaseIndex + 1];
      
      await this.prisma.mission.update({
        where: { id: missionId },
        data: { phase: nextPhase.name }
      });

      await this.diana.sendPhaseAdvancementMessage(
        mission.userId,
        missionId,
        nextPhase
      );
    }
  }

  // 5. Complete mission
  async completeMission(missionId: string): Promise<void> {
    await this.prisma.mission.update({
      where: { id: missionId },
      data: {
        status: 'COMPLETED',
        completionDate: new Date(),
        progress: 100
      }
    });

    // Send celebration message
    const mission = await this.prisma.mission.findUnique({
      where: { id: missionId }
    });
    
    await this.diana.sendMissionCompletionMessage(mission.userId, mission);
  }
}
```

// apps/api/src/chat/chat.controller.ts (Mission-aware)

@Post('chat/stream')
async streamChat(
  @Body() body: ChatStreamDto,
  @Req() req: Request
): Promise<StreamResponse> {
  const userId = req.user.id;
  
  // Check if user has active mission
  const activeMission = await this.missionService.getActiveMission(userId);
  
  // Build context with mission awareness
  const context = await this.contextEngine.buildMessages(
    userId,
    body.conversationId,
    { 
      activeMission,
      missionPhase: activeMission?.phase,
      missionDocuments: activeMission?.documents
    }
  );

  // Diana is mission-aware
  const systemPrompt = `
    User's current mission: ${activeMission?.name}
    Phase: ${activeMission?.phase}
    
    When responding, reference the mission context.
    Suggest next steps in the mission roadmap.
    Update progress as they work.
  `;

  // Stream response
  for await (const chunk of this.llmService.stream(context.messages)) {
    res.write(`data: ${JSON.stringify(chunk)}\n\n`);
  }
}
